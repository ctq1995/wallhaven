    - Implementation:
      ```typescript
      export function getAppSetting(key: string): unknown {
        try {
          const row = getDatabase()
            .prepare<{ value: string }>('SELECT value FROM settings WHERE key = ?')
            .get(key)
          if (!row) return null
          return JSON.parse(row.value)
        } catch {
          return null
        }
      }
      ```
    - This uses parameterized query (? placeholder) to prevent SQL injection (ASVS V5)
    - Returns null on any error (key not found, parse failure, DB not accessible)
    - Uses getDatabase() which triggers lazy initialization if not already open

      export function getDownloadPath(): string | undefined {

        if (appSettings && typeof appSettings.downloadPath === 'string') {
          return appSettings.downloadPath
        }
        return undefined
