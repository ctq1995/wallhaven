5. 页码计算逻辑示例:
```typescript
const visiblePages = computed(() => {
  if (props.totalPages <= 5) {
    // 总页数 ≤ 5，全部显示
    return Array.from({ length: props.totalPages }, (_, i) => i + 1)
  }

  // 计算显示范围
  let start = Math.max(1, props.currentPage - 2)
  let end = Math.min(props.totalPages, props.currentPage + 2)

  // 边界自适应
  if (props.currentPage <= 3) {
    end = 5
  } else if (props.currentPage >= props.totalPages - 2) {
    start = props.totalPages - 4
  }

  return Array.from({ length: end - start + 1 }, (_, i) => start + i)
})

const showStartEllipsis = computed(() => {
  return props.totalPages > 5 && props.currentPage > 3
})

const showEndEllipsis = computed(() => {
  return props.totalPages > 5 && props.currentPage < props.totalPages - 2
})
```

6. 模板结构 (复用 list.css 样式):
```html
<nav class="pagination">
  <ul>
    <!-- 上一页按钮 -->
    <li :class="{ disabled: props.currentPage <= 1 || props.loading }">
      <a v-if="props.currentPage > 1 && !props.loading"
         href="#"
         @click.prevent="emit('go-to-page', props.currentPage - 1)">
        上一页
      </a>
      <span v-else>上一页</span>
    </li>

    <!-- 首页 -->
    <li v-if="props.totalPages > 1"
        :class="{ active: props.currentPage === 1 }">
      <a href="#" @click.prevent="handlePageClick(1)">1</a>
    </li>

    <!-- 左侧省略号 -->
    <li v-if="showStartEllipsis"><span>...</span></li>

    <!-- 中间页码 -->
    <li v-for="page in visiblePages"
        :key="page"
        v-show="page !== 1 && page !== props.totalPages"
        :class="{ active: page === props.currentPage }">
      <a href="#" @click.prevent="handlePageClick(page)">{{ page }}</a>
    </li>

    <!-- 右侧省略号 -->
    <li v-if="showEndEllipsis"><span>...</span></li>

    <!-- 末页 -->
    <li v-if="props.totalPages > 1"
        :class="{ active: props.currentPage === props.totalPages }">
      <a href="#" @click.prevent="handlePageClick(props.totalPages)">
        {{ props.totalPages }}
      </a>
    </li>

    <!-- 下一页按钮 -->
    <li :class="{ disabled: props.currentPage >= props.totalPages || props.loading }">
      <a v-if="props.currentPage < props.totalPages && !props.loading"
         href="#"
         @click.prevent="emit('go-to-page', props.currentPage + 1)">
        下一页
      </a>
      <span v-else>下一页</span>
    </li>
  </ul>

  <!-- 总条目数 -->
  <span class="pagination-notice">
    共 {{ formatCount(props.totalCount) }} 张
  </span>
</nav>
```

7. 辅助函数:
```typescript
// 格式化总条目数（添加千分位）
function formatCount(count: number): string {

// 处理页码点击
function handlePageClick(page: number): void {
  if (props.loading || page === props.currentPage) return

}
```

```typescript
/**
 * 更新当前页中某个壁纸的收藏状态
 * @param wallpaperId - 壁纸 ID
 * @param isFavorite - 新的收藏状态值 (0, 1, 2)
 */
function updateItemFavoriteStatus(wallpaperId: string, isFavorite: number): void {
  const currentData = store.currentPageData
  const itemIndex = currentData.data.findIndex(item => item.id === wallpaperId)

  if (itemIndex === -1) return

  // 创建新的数据对象以触发 shallowRef 更新
  const newData = [...currentData.data]
  newData[itemIndex] = { ...newData[itemIndex], is_favorite: isFavorite }

  store.currentPageData = {
    ...currentData,
    data: newData
  }

  // 同时更新缓存
  const page = currentData.currentPage
  const cachedPage = store.getCachedPage(page)
  if (cachedPage) {
    const cachedData = [...cachedPage.data]
    const cachedIndex = cachedData.findIndex(item => item.id === wallpaperId)
    if (cachedIndex !== -1) {
      cachedData[cachedIndex] = { ...cachedData[cachedIndex], is_favorite: isFavorite }
      store.setCachedPage(page, { ...cachedPage, data: cachedData })
    }
  }
}
```

```typescript
/**
 * 键盘导航处理（与 ImagePreview 互斥）
 */
const handleKeydown = (event: KeyboardEvent): void => {
  // 只在 ImagePreview 关闭时响应
  if (imgShow.value) return

  const { currentPage, totalPage } = currentPageData.value

  // 边界检查 + 导航
  if (event.key === 'ArrowLeft' && currentPage > 1) {
    event.preventDefault()
    goToPage(currentPage - 1)
  } else if (event.key === 'ArrowRight' && currentPage < totalPage) {
    event.preventDefault()
    goToPage(currentPage + 1)
  }
}
```

2. 在 onMounted 中添加事件监听:
```typescript
onMounted(() => {
  document.addEventListener('click', handleClickOutside)
  window.addEventListener('keydown', handleKeydown)  // 新增

```

3. 在 onUnmounted 中移除事件监听:
```typescript
onUnmounted(() => {
  // 移除滚动监听（已在 Task 3 移除）
  document.removeEventListener('click', handleClickOutside)
  window.removeEventListener('keydown', handleKeydown)  // 新增
```

**互斥机制说明:**
- ImagePreview: `if (!props.showing) return` — 只在预览开启时响应
- OnlineWallpaper: `if (imgShow.value) return` — 只在预览关闭时响应
